# This script enables a CCTV camera an identifies the face using OpenCV
# It then saves the face and enhances the image

import pathlib
import cv2

cascade_path = pathlib.Path(cv2.__file__).parent.absolute() / "data/haarcascade_frontalface_default.xml"

clf = cv2.CascadeClassifier(str(cascade_path))

camera = cv2.VideoCapture(0)

while True:
    _, frame = camera.read()
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    faces = clf.detectMultiScale(
        gray,
        scaleFactor=1.1,
        minNeighbors=5,
        minSize=(30, 30),
        flags=cv2.CASCADE_SCALE_IMAGE
    )

    # show the face detection in live feed
    # for (x, y, width, height) in faces:
    # cv2.rectangle(frame, (x+150, y-50), (x-width+150, y+height+50), (0, 0, 255), 1)  #red

    count = 0
    for (x, y, w, h) in faces:
        face = frame[y:y + h, x:x + w]  # slice the face from the image
        cv2.imwrite(str(count) + '.jpg', face)  # save the image
        count += 1
        cv2.rectangle(frame, (x, y), (x + w, y + h), (255, 0, 0), 2) #blue

# close the CCTV window using 'q'
    cv2.imshow("Faces", frame)
    if cv2.waitKey(1) == ord("q"):
        break

camera.release()
cv2.destroyAllWindows()
